\section*{Conventions}
% ============================================================

All cases share these conventions unless explicitly overridden:
\begin{itemize}
  \item $\gamma = 1.0$ (no discounting), to keep arithmetic clean.
  \item $B_1 = 3 \cdot \sum_i \BAC_i$ (large; cash never binding unless stated).
  \item $\bar\eta_{i,t} = \bar\eta$ constant and deterministic unless stated.
  \item Equal milestone weights $\phi_{i,j} = 1/M_i$ unless stated.
  \item $\alpha_i = 0$, $\rho_i = 0$ (no advance, no retention) unless stated.
  \item $\delta^{\mathrm{rec}} = 0.25$, $\psi = 0.10$ (FIDIC defaults) unless stated.
  \item $\pi_i = 0$ unless stated ($CP_i = \BAC_i$).
  \item S-curve shape $a_i = b_i = 2$ (symmetric Beta) unless stated.
  \item Monetary units are abstract.
\end{itemize}

\section{Group 1: Progress and Milestone Certification}
% ============================================================

\textit{What is being proven:} budget converts to progress at rate $\eta$,
milestones certify at the correct threshold, payments trigger correctly, and
milestone ordering is enforced even when progress jumps past multiple
thresholds in a single period.

% ------------------------------------------------------------
\subsection{SP-1: Perfect Efficiency, Two Milestones, Front-Load}
% ------------------------------------------------------------

\paragraph{Economic claim.}
With $\eta=1$, no budget pressure, and $\gamma=1$, the MILP front-loads
spend to certify milestones as early as possible. No termination conditions
fire.

\paragraph{Parameters.}
\[
  n=1,\quad s_1=1,\quad f_1=4,\quad \BAC_1=100,\quad CP_1=120,\quad \pi_1=0.20.
\]
\[
  M_1=2,\quad \theta=(0.5,\;1.0),\quad \phi=(0.5,\;0.5),\quad \bar\eta=1.0.
\]
\[
  \alpha_1=0,\quad \rho_1=0,\quad \Omega_1=1,\quad \mu_1=0.30,\quad
  \tau_1^{\mathrm{tol}}=2.
\]
\[
  B_1=300,\quad H=4.
\]

\paragraph{Milestone payments.}
\[
  R_{1,1}^{\mathrm{net}} = 0.5\times120 = 60,\qquad
  R_{1,2}^{\mathrm{net}} = 0.5\times120 = 60.
\]
Identity: $0+60+60+0=120=CP_1$. \checkmark

\paragraph{Optimal strategy.}
Spend 50 at $t=1$ (MS1 certified), then 50 at $t=2$ (MS2 certified).
\begin{center}
\begin{tabular}{cccccc}
\toprule
$t$ & $x_{1,t}$ & $P_{1,t}$ & Inflow & Net & $B_t$\\
\midrule
1 & 50 & 0.50 & 60 & $+10$ & $300\to310$\\
2 & 50 & 1.00 & 60 & $+10$ & $310\to320$\\
3 &  0 & 1.00 &  0 & $0$   & $320$\\
4 &  0 & 1.00 &  0 & $0$   & $320$\\
\bottomrule
\end{tabular}
\end{center}

\paragraph{Objective.} $Z^*_{\mathrm{L1}} = (60-50)+(60-50) = \mathbf{20}$.

\paragraph{Termination check.}
$\CPI=1.0$, $\EAC=100\leq1.30\times100$. Neither condition fires. \checkmark

\paragraph{Assertions for pytest.}
\begin{enumerate}[label=A\arabic*.]
  \item $x_{1,1}=50\pm0.5$, $x_{1,2}=50\pm0.5$, $x_{1,3}=x_{1,4}=0$.
  \item $P_{1,1}=0.50\pm0.01$, $P_{1,2}=1.00\pm0.01$.
  \item MS1 certified at $t=1$; MS2 certified at $t=2$.
  \item Status: \texttt{completed}. No $\tau$ decrement.
  \item $|Z^*-20|\leq1.0$.
\end{enumerate}

\section{Group 2: Discounting and Timing Incentives}
% ============================================================

\textit{What is being proven:} $\gamma<1$ creates a genuine preference for
earlier milestone certification; the MILP exploits this by front-loading
allocations to projects whose milestone structure yields higher discounted
inflows when completed first.

% ------------------------------------------------------------
\subsection{SP-4: Front-Loading Beats Uniform Allocation ($\gamma<1$)}
% ------------------------------------------------------------

\paragraph{Economic claim.}
With discounting, earlier inflows are worth more. The MILP front-loads
all spending to certify milestones at $t=1$ and $t=2$, beating the
uniform allocation by a computable margin.

\paragraph{Parameters.}
\[
  n=1,\quad s_1=1,\quad f_1=8,\quad \BAC_1=100,\quad\pi_1=0.40
  \Rightarrow CP_1=140.
\]
\[
  M_1=2,\quad\theta=(0.5,\;1),\quad\phi=(0.5,\;0.5),\quad
  \bar\eta=1.0,\quad\gamma=0.95.
\]
\[
  \alpha_1=0,\quad\rho_1=0,\quad R_{1,j}^{\mathrm{net}}=70\text{ each.}
\]

\paragraph{Front-loaded vs.\ uniform.}
\[
  Z^{\mathrm{front}} = 70+0.95\times70-50-0.95\times50
  = 70+66.50-50-47.50 = \mathbf{39.00}.
\]
Uniform (MS1 at $t=4$, MS2 at $t=8$):
\[
  Z^{\mathrm{uniform}}
  = 0.95^3\times70+0.95^7\times70-12.5\sum_{t=0}^{7}0.95^t
  \approx 60.02+48.88-80.29 = \mathbf{28.61}.
\]

\paragraph{Assertions for pytest.}
\begin{enumerate}[label=A\arabic*.]
  \item $x_{1,1}=50\pm0.5$, $x_{1,2}=50\pm0.5$.
  \item MS1 certified at $t=1$; MS2 certified at $t=2$.
  \item $Z^*\geq38.5$.
  \item $Z^*>28.61$.
\end{enumerate}

% ------------------------------------------------------------
\subsection{MP-K: Two Projects, $\gamma<1$, Front-Heavy vs.\ Back-Heavy Milestones}
% ------------------------------------------------------------

\paragraph{Economic claim.}
Two projects are identical in $\BAC$, $CP$, $\eta$, and duration but differ
in milestone weight distribution. Under $\gamma<1$, the MILP completes the
front-heavy project (large early payment) before the back-heavy project,
since completing it first yields higher discounted total inflow.

\paragraph{Parameters.}
\[
  n=2,\quad s_i=1,\quad f_i=6,\quad\BAC_i=60,\quad\pi_i=0.50
  \Rightarrow CP_i=90.
\]
\[
  M_i=2,\quad\theta_i=(0.5,\;1.0),\quad\bar\eta_i=1.0,\quad\gamma=0.90.
\]
\[
  \phi_A=(0.70,\;0.30),\quad\phi_B=(0.30,\;0.70).
\]
\[
  \alpha_i=0,\quad\rho_i=0,\quad B_1=300.
\]

\paragraph{Milestone payments.}
Project A: $R_{A,1}^{\mathrm{net}}=63$, $R_{A,2}^{\mathrm{net}}=27$.
Project B: $R_{B,1}^{\mathrm{net}}=27$, $R_{B,2}^{\mathrm{net}}=63$.
Each milestone requires spend of 30.

\paragraph{Strategy comparison.}
\textit{Strategy 1 (A first, then B):}
\[
  Z = (63-30)+0.90(27-30)+0.81(27-30)+0.729(63-30)
    = 33-2.7-2.43+24.06 = \mathbf{51.93}.
\]
\textit{Strategy 2 (B first, then A):}
\[
  Z = (27-30)+0.90(63-30)+0.81(63-30)+0.729(27-30)
    = -3+29.7+26.73-2.19 = \mathbf{51.24}.
\]
Strategy~1 dominates. MILP must choose A first or any strategy with
$Z\geq51.93$.

\paragraph{Assertions for pytest.}
\begin{enumerate}[label=A\arabic*.]
  \item Project A completed before or simultaneously with Project B.
  \item $Z^*\geq51.90$.
  \item $Z^*>51.24$ (B-first strategy strictly dominated).
  \item Both projects \texttt{completed}.
\end{enumerate}

\section{Group 3: Advance Payment and Recovery Mechanics}
% ============================================================

\textit{What is being proven:} advance payment arrives at $t=s_i$,
recovery deductions apply correctly once the cumulative payment threshold
$\psi_i$ is crossed, deductions stop once the advance is fully recovered,
and the payment identity $A_i+\sum_j R_{i,j}^{\mathrm{net}}+R_i^{\mathrm{ret}}=CP_i$
holds exactly.

% ------------------------------------------------------------
\subsection{SP-2: Advance Harvest, Efficiency Decay, Forced Termination}
% ------------------------------------------------------------

\paragraph{Economic claim.}
A large advance combined with rapidly decaying efficiency makes the project
economically dead within two periods. The MILP receives the advance and
accepts termination. The settlement exactly cancels the advance, yielding
$Z^*=0$.

\paragraph{Parameters.}
\[
  n=1,\quad s_1=1,\quad f_1=6,\quad\BAC_1=100,\quad CP_1=100,\quad\pi_1=0.
\]
\[
  M_1=2,\quad\theta=(0.5,\;1.0),\quad\phi=(0.5,\;0.5).
\]
\[
  \alpha_1=0.30\Rightarrow A_1=30,\quad\delta^{\mathrm{rec}}=0.25,\quad
  \psi=0.10,\quad\rho_1=0.
\]
\[
  \Omega_1=1,\quad\mu_1=0.05,\quad\tau_1^{\mathrm{tol}}=1.
\]
\[
  \bar\eta_{1,1}=1.00,\quad\bar\eta_{1,t}=0.05\text{ for }t\geq2.
\]

\paragraph{Termination and settlement.}
With zero spend from $t=1$: counter $1\to0$ at $t=1$, termination at $t=2$.
$P_{1,2}=0$. $R_1^{\mathrm{term}}=0-30=-30$.
$Z^*=30+0-30=\mathbf{0}$.

\paragraph{Assertions for pytest.}
\begin{enumerate}[label=A\arabic*.]
  \item $\sum_t x_{1,t}\leq1.0$.
  \item Project terminates at $t\leq3$.
  \item $R_1^{\mathrm{term}}=-30\pm1$.
  \item No milestone certified.
  \item $|Z^*-0|\leq1.5$.
\end{enumerate}

\subsection{SP-A: Late Recovery Threshold ($\psi=0.40$)}
% ------------------------------------------------------------

\paragraph{Economic claim.}
When $\psi=0.40$, the cumulative certified payment fraction must exceed 40\%
before recovery deductions begin. MS1 carries weight $\phi_1=1/3<0.40$, so
its payment is received in full. Deductions apply from MS2 onward. The
payment identity holds exactly.

\paragraph{Parameters.}
\[
  n=1,\quad s_1=1,\quad f_1=6,\quad\BAC_1=120,\quad\pi_1=0.50
  \Rightarrow CP_1=180.
\]
\[
  M_1=3,\quad\theta=(1/3,\;2/3,\;1),\quad\phi=(1/3,\;1/3,\;1/3).
\]
\[
  \alpha_1=0.20\Rightarrow A_1=36,\quad\psi=0.40,\quad
  \delta^{\mathrm{rec}}=0.30,\quad\rho_1=0,\quad\bar\eta=1.0.
\]
\[
  \Omega_1=2,\quad\mu_1=0.50,\quad\tau_1^{\mathrm{tol}}=2,\quad B_1=400.
\]

\paragraph{Recovery threshold check.}
\begin{align*}
  \text{After MS1:}&\quad\textstyle\sum_{k\leq1}\phi_k=1/3\approx0.333<0.40
    \Rightarrow\text{no deduction.}\\
  \text{After MS2:}&\quad\textstyle\sum_{k\leq2}\phi_k=2/3\approx0.667\geq0.40
    \Rightarrow\text{deduction applies.}\\
  \text{After MS3:}&\quad\textstyle\sum_{k\leq3}\phi_k=1.0\geq0.40
    \Rightarrow\text{deduction applies.}
\end{align*}

\paragraph{Net payments.}
Each milestone gross $=60$.
\begin{align*}
  R_{1,1}^{\mathrm{net}} &= 60 - 0 = 60.\\
  R_{1,2}^{\mathrm{net}} &= 60 - 0.30\times60 = 60-18 = 42.\\
  R_{1,3}^{\mathrm{net}} &= 60 - 0.30\times60 = 60-18 = 42.
\end{align*}
Total deductions $=36=A_1$. Identity: $36+60+42+42=180=CP_1$. \checkmark

\paragraph{Allocation.}
Spend $x_t=40$ per period; MS$j$ certifies at $t=j$.

\paragraph{Objective.}
$Z^*=(36+60-40)+(42-40)+(42-40)=56+2+2=\mathbf{60}$.

\begin{notebox}
\NOTEH MS1 payment is received in full (no deduction) because cumulative
weight $1/3<\psi=0.40$. This isolates the threshold mechanic from the
SP-2 case where deductions begin at the first milestone.
\end{notebox}

\paragraph{Assertions for pytest.}
\begin{enumerate}[label=A\arabic*.]
  \item $R_{1,1}^{\mathrm{net}}=60\pm0.5$ (no deduction).
  \item $R_{1,2}^{\mathrm{net}}=42\pm0.5$ (deduction applies).
  \item $R_{1,3}^{\mathrm{net}}=42\pm0.5$ (deduction applies).
  \item Total inflow $=180=CP_1$.
  \item $|Z^*-60|\leq1.5$.
\end{enumerate}

% ------------------------------------------------------------

\paragraph{Economic claim.}
When recovery deductions on MS1 and MS2 exactly exhaust the advance,
MS3 carries no deduction despite $\delta^{\mathrm{rec}}>0$. The final
payment equals the gross milestone amount.

\paragraph{Parameters.}
\[
  n=1,\quad s_1=1,\quad f_1=4,\quad\BAC_1=60,\quad\pi_1=0.50
  \Rightarrow CP_1=90.
\]
\[
  M_1=3,\quad\theta=(1/3,\;2/3,\;1),\quad\phi=(1/3,\;1/3,\;1/3).
\]
\[
  \alpha_1=0.20\Rightarrow A_1=18,\quad\psi=0.10,\quad

\[
  \Omega_1=2,\quad\mu_1=0.50,\quad\tau_1^{\mathrm{tol}}=2,\quad B_1=200.

$Z^*=(18+21-20)+(21-20)+(30-20)=19+1+10=\mathbf{30}$.

\paragraph{Assertions for pytest.}
\begin{enumerate}[label=A\arabic*.]
  \item $R_{1,1}^{\mathrm{net}}=21\pm0.5$.
  \item $R_{1,2}^{\mathrm{net}}=21\pm0.5$.
  \item $R_{1,3}^{\mathrm{net}}=30\pm0.5$ (no deduction on final milestone).
  \item Total inflow $=90=CP_1$.
  \item $|Z^*-30|\leq1.0$.
\end{enumerate}

\section{Group 4: Retention}
% ============================================================

\textit{What is being proven:} retention fraction $\rho_i$ is withheld
from every milestone payment, accumulates to $\rho_i\cdot CP_i$, and is
released as a single lump sum at project completion only. Retention and
advance recovery interact correctly when both are active.

% ------------------------------------------------------------
\subsection{SP-C: Retention Only, Lump Release at Completion}
% ------------------------------------------------------------

\paragraph{Economic claim.}
With $\rho=0.10$ and no advance, each milestone payment is reduced by
10\%. The withheld amounts accumulate and are released in full when the
final milestone is certified.

\paragraph{Parameters.}
\[
  n=1,\quad s_1=1,\quad f_1=6,\quad\BAC_1=60,\quad\pi_1=0.50
  \Rightarrow CP_1=90.
\]
\[
  M_1=3,\quad\theta=(1/3,\;2/3,\;1),\quad\phi=(1/3,\;1/3,\;1/3),\quad\bar\eta=1.0.
\]
\[
  \alpha_1=0,\quad\rho_1=0.10,\quad\Omega_1=2,\quad\mu_1=0.50,\quad
  \tau_1^{\mathrm{tol}}=2.
\]

\paragraph{Net payments.}
Each gross milestone $=30$. Retention per milestone $=0.10\times30=3$.
\[
  R_{1,j}^{\mathrm{net}}=30\times(1-0.10)=27\quad\text{for }j=1,2,3.
\]
Retention release at completion: $R_1^{\mathrm{ret}}=0.10\times90=9$.

Final period inflow $= R_{1,3}^{\mathrm{net}}+R_1^{\mathrm{ret}}=27+9=36$.

Identity: $0+27+27+27+9=90=CP_1$. \checkmark

\paragraph{Objective.}
Spend $x_t=20$ per period.
$Z^*=(27+27+36)-(20+20+20)=90-60=\mathbf{30}$.

\paragraph{Assertions for pytest.}
\begin{enumerate}[label=A\arabic*.]
  \item $R_{1,j}^{\mathrm{net}}=27\pm0.5$ for $j=1,2,3$.
  \item $R_1^{\mathrm{ret}}=9\pm0.5$ released at final certification only.
  \item No retention released before project \texttt{completed}.
  \item Total inflow $=90=CP_1$.
  \item $|Z^*-30|\leq1.0$.
\end{enumerate}

% ------------------------------------------------------------
\subsection{SP-D: Retention and Advance Recovery Simultaneously Active}
% ------------------------------------------------------------

\paragraph{Economic claim.}
When both $\rho>0$ and $\alpha>0$, each milestone payment is reduced by
both retention withholding and advance recovery deduction. After the
advance is fully recovered, only retention applies. The payment identity
holds exactly.

\paragraph{Parameters.}
\[
  n=1,\quad s_1=1,\quad f_1=6,\quad\BAC_1=60,\quad\pi_1=0.50
  \Rightarrow CP_1=90.
\]
\[
  M_1=3,\quad\theta=(1/3,\;2/3,\;1),\quad\phi=(1/3,\;1/3,\;1/3),\quad\bar\eta=1.0.
\]
\[
  \alpha_1=0.20\Rightarrow A_1=18,\quad\psi=0.10,\quad
  \delta^{\mathrm{rec}}=0.30,\quad\rho_1=0.10.
\]
\[
  \Omega_1=2,\quad\mu_1=0.50,\quad\tau_1^{\mathrm{tol}}=2,\quad B_1=200.
\]

\paragraph{Payment mechanics.}
Each gross milestone $=30$. Retention per milestone $=3$.
\begin{align*}
  R_{1,1}^{\mathrm{net}} &= 30\times0.90 - 0.30\times30 = 27-9=18.
    \quad\text{Recovered: }9.\\
  R_{1,2}^{\mathrm{net}} &= 27-9=18.
    \quad\text{Recovered: }18=A_1.\quad\text{Advance fully recovered.}\\
  R_{1,3}^{\mathrm{net}} &= 27-0=27.
    \quad\text{Only retention; no recovery deduction.}
\end{align*}
$R_1^{\mathrm{ret}}=0.10\times90=9$ at completion.

Identity: $18+18+18+27+9=90=CP_1$. \checkmark

\paragraph{Objective.}
$Z^*=(18+18+18+27+9)-(20+20+20)=90-60=\mathbf{30}$.

\paragraph{Assertions for pytest.}
\begin{enumerate}[label=A\arabic*.]
  \item $R_{1,1}^{\mathrm{net}}=18\pm0.5$ (both retention and recovery deducted).
  \item $R_{1,2}^{\mathrm{net}}=18\pm0.5$ (both deducted; advance now recovered).
  \item $R_{1,3}^{\mathrm{net}}=27\pm0.5$ (retention only; no recovery deduction).
  \item $R_1^{\mathrm{ret}}=9\pm0.5$ at completion.
  \item Total inflow $=90=CP_1$.
  \item $|Z^*-30|\leq1.0$.
\end{enumerate}

% ============================================================
\section{Group 5: Termination Conditions in Isolation}
% ============================================================

\textit{What is being proven:} Conditions~1 and~2 are independent. The
cure-period counter decrements only when \emph{both} conditions hold
simultaneously. When only one holds, the counter must remain at
$\tau_i^{\mathrm{tol}}$. When one lapses mid-cure, the counter resets.

\begin{notebox}
\NOTEH A key identity: $\CPI_{i,t}=\eta_{i,t}$ always. Therefore
$\CPI=1$ if and only if $\eta=1$, and $\EAC=\BAC$ whenever $\eta=1$,
making Condition~2 structurally impossible to fire under $\eta=1$.
\end{notebox}

% ------------------------------------------------------------
\subsection{SP-E: Only Condition~1 Fires (Schedule Lag, CPI~$=$~1)}
% ------------------------------------------------------------

\paragraph{Economic claim.}

conditions fire from $t=1$ (with zero spend, SPI undefined; default to
termination). Lo terminates; $A_{\mathrm{Lo}}=0$, $R_{\mathrm{Lo}}^{\mathrm{term}}=0$.

$Z^*=Z_{\mathrm{Hi}}+0=\mathbf{30}$.

\paragraph{Assertions for pytest.}
\begin{enumerate}[label=A\arabic*.]
  \item $\sum_t x_{\mathrm{Lo},t}\leq1.0$.
  \item Lo: \texttt{terminated}.
  \item Hi: \texttt{completed}.
  \item $|Z^*-30|\leq1.5$.
  \item $B_t\geq0$ for all $t$.
\end{enumerate}

\subsection{MP-D: Early Milestone Cash Unlocks Higher-NPV Project}
% ------------------------------------------------------------

\paragraph{Economic claim.}
A low-NPV project with an early milestone can be worth funding partially
if its payment replenishes the balance sufficiently to enable completion
of a higher-NPV project that would otherwise stall. The MILP evaluates
this option correctly.

\paragraph{Parameters.}
\[
  n=3,\quad s_i=1,\quad B_1=40.
\]
\textit{Project Bridge:} $f=3$, $\BAC=30$, $\pi=0.10\Rightarrow CP=33$,
$M=1$, $\theta=(1.0)$, $\phi=(1.0)$, $\bar\eta=1.0$, $\alpha=0$,
$\rho=0$, $\Omega=2$, $\mu=0.50$, $\tau^{\mathrm{tol}}=2$.

\textit{Project Alpha:} $f=6$, $\BAC=60$, $\pi=0.50\Rightarrow CP=90$,
$M=2$, $\theta=(0.5,1)$, $\phi=(0.5,0.5)$, $\bar\eta=1.0$, $\alpha=0$,
$\rho=0$, $\Omega=2$, $\mu=0.50$, $\tau^{\mathrm{tol}}=2$.

\textit{Project Beta:} same as Alpha.

\paragraph{Strategy 1: fund Bridge first.}
$t=1$: spend 30 on Bridge. $B_2=40-30+33=43$.
$t=2$: spend 30 on Alpha (MS1, $R=45$), 13 on Beta. $B_3=43-43+45=45$.
$t=3$: spend 17 on Beta (cum.\ 30, MS1$_\beta$, $R=45$), 28 on Alpha.
$B_4=45-45+45=45$.
$t=4$: spend 30 on Alpha (MS2, $R=45$), 15 on Beta. $B_5=45-45+45=45$.
$t=5$: spend 15 on Beta (MS2, $R=45$). $B_6=45-15+45=75$.

This gets complex. The key economic point is simpler to state:
$Z_{\mathrm{Bridge}}=33-30=3$ (small positive).
If Bridge enables completion of Alpha and Beta which otherwise stall,
the total $Z=3+30+30=63$ versus not funding Bridge and completing only
one of Alpha/Beta with $Z=30$.
MILP must recognize the option value of Bridge.

\paragraph{Assertions for pytest.}
\begin{enumerate}[label=A\arabic*.]
  \item Bridge funded at $t=1$: $x_{\mathrm{Bridge},1}\geq28$.
  \item Bridge \texttt{completed}.
  \item $Z^*>30$ (MILP does better than completing only one major project).
  \item $B_t\geq0$ for all $t$.
\end{enumerate}

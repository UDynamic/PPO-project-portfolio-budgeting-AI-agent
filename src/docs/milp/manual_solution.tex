\section{Hand-Solved MILP Test Cases}
% ============================================================

All cases share these conventions unless explicitly relaxed:
\begin{itemize}
  \item $\gamma = 1.0$ (no discounting), to keep arithmetic clean.
  \item $B_1 = 3 \cdot \sum_i \BAC_i$ (large; cash never binding).
  \item $\bar\eta_{i,t} = \bar\eta$ constant and deterministic.
  \item Equal milestone weights: $\phi_{i,j} = 1/M_i$.
  \item $\alpha_i = 0$, $\rho_i = 0$ (no advance, no retention) unless stated.
  \item $\delta^{\mathrm{rec}} = 0.25$, $\psi = 0.10$ (FIDIC defaults).
  \item $\pi_i = 0$ unless stated ($CP_i = \BAC_i$: baseline with zero margin).
  \item Monetary units are abstract (``k'').
  \item S-curve shape: $a_i = b_i = 2$ (symmetric Beta) throughout.
\end{itemize}

\subsubsection{SP-1: Perfect Efficiency, No Termination Risk}
% ------------------------------------------------------------

\paragraph{Economic claim.}
With $\eta=1$, no budget pressure, and $\gamma=1$, the MILP front-loads spend
to certify milestones as early as possible. No termination conditions fire.

\paragraph{Parameters.}
\[
  n=1,\quad s_1=1,\quad f_1=4,\quad \BAC_1=100,\quad CP_1=120,\quad
  \pi_1=0.20.
\]
\[
  M_1=2,\quad \theta=(0.5,\;1.0),\quad \phi=(0.5,\;0.5),\quad
  \bar\eta=1.0.
\]
\[
  \alpha_1=0,\quad \rho_1=0,\quad
  \Omega_1=1,\quad \mu_1=0.30,\quad \tau_1^{\mathrm{tol}}=2.
\]
\[
  B_1=300,\quad H=4.
\]

\paragraph{Milestone payments.}
\[
  R_{1,1}^{\mathrm{net}} = 0.5 \times 120 = 60,\qquad
  R_{1,2}^{\mathrm{net}} = 0.5 \times 120 = 60.
\]
Payment identity: $0 + 60 + 60 + 0 = 120 = CP_1$. \checkmark

\paragraph{Progress mechanics.}
$P_{1,t} = \eta \cdot x_{1,t} / \BAC_1 = x_{1,t}/100$ per period.
To certify MS1, need cumulative $P \geq 0.5$, i.e.\ cumulative spend $\geq 50$.
To certify MS2, need cumulative $P = 1$, i.e.\ cumulative spend $= 100$.

\paragraph{Optimal strategy.}
With $\gamma=1$ and cash non-binding, the MILP maximises inflow speed.
Spend 50 at $t=1$ (MS1 certified immediately), then 50 at $t=2$ (MS2 certified).

\begin{center}
\begin{tabular}{cccccc}
\toprule
$t$ & $x_{1,t}$ & $P_{1,t}$ & Inflow & Net & $B_t$\\
\midrule
1 & 50 & 0.50 & 60 (MS1) & $+10$ & $300\to310$\\
2 & 50 & 1.00 & 60 (MS2) & $+10$ & $310\to320$\\
3 &  0 & 1.00 &  0       &   $0$ & $320$\\
4 &  0 & 1.00 &  0       &   $0$ & $320$\\
\bottomrule
\end{tabular}
\end{center}

\paragraph{Objective.}
\[
  Z^*_{\mathrm{L1}} = (60-50)+(60-50) = \mathbf{20}.
\]

\paragraph{Termination check.}
At $t=1$: $P^{\mathrm{plan}}_{1,1} = I_{0.25}(2,2) \approx 0.156$.
$\SPI = 0.50/0.156 = 3.2 \gg 1$. $\CPI = 50/50 = 1.0$.
$\EAC = 50 + 50/1 = 100 \leq 1.30 \times 100$. Neither condition fires. \checkmark

\paragraph{Assertions for pytest.}
\begin{enumerate}[label=A\arabic*.]
  \item $x_{1,1} = 50\pm 0.5$, $x_{1,2} = 50\pm 0.5$, $x_{1,3}=x_{1,4}=0$.
  \item $P_{1,1} = 0.50\pm 0.01$, $P_{1,2} = 1.00\pm 0.01$.
  \item MS1 certified at $t=1$; MS2 certified at $t=2$.
  \item Status: \texttt{completed}. No $\tau$ decrement.
  \item $|Z^* - 20| \leq 1.0$.
\end{enumerate}

\subsubsection{SP-2: Advance Payment Harvesting, Forced Termination}
% ------------------------------------------------------------

\paragraph{Economic claim.}
A large advance combined with rapidly decaying efficiency makes the project
economically dead within two periods. The MILP receives the advance and accepts
termination. The settlement exactly cancels the advance, yielding $Z^*=0$.

\paragraph{Parameters.}
\[
  n=1,\quad s_1=1,\quad f_1=6,\quad \BAC_1=100,\quad CP_1=100,\quad \pi_1=0.
\]
\[
  M_1=2,\quad \theta=(0.5,\;1.0),\quad \phi=(0.5,\;0.5).
\]
\[
  \alpha_1=0.30 \Rightarrow A_1=30,\quad
  \delta^{\mathrm{rec}}=0.25,\quad \psi=0.10,\quad \rho_1=0.
\]
\[
  \Omega_1=1,\quad \mu_1=0.05,\quad \tau_1^{\mathrm{tol}}=1.
\]
Efficiency profile (geometric decay):
\[
  \bar\eta_{1,t} = \begin{cases}
    1.00 & t=1\\
    0.05 & t\geq 2
  \end{cases}
\]

\paragraph{Milestone payment computation.}
Recovery threshold $\psi=0.10$; MS1 weight $\phi_1=0.5 \geq 0.10$, so recovery
starts at MS1.
\begin{align*}
  R_{1,1}^{\mathrm{net}} &= 0.5\times100 - 0.25\times0.5\times100 = 50-12.5=37.5\\
  R_{1,2}^{\mathrm{net}} &= 0.5\times100 - 0.25\times0.5\times100 = 37.5
\end{align*}
Total recovered via deductions $= 25$. Advance outstanding at full completion $= 5$.
(The remaining 5 is recovered at final settlement; identity: $30 + 37.5 + 37.5 - 5 = 100$.)

\paragraph{Feasibility of reaching MS1 from $t=2$.}
To reach $\theta=0.5$ starting from $P=0$ with $\eta=0.05$:
\[
  x^{\mathrm{needed}} = 0.5 \times \BAC / \eta = 0.5\times100/0.05 = 1000 \gg \BAC.
\]
\textbf{MS1 is physically unreachable from $t=2$.}

\paragraph{Termination under zero spend.}
\begin{itemize}
  \item $t=1$, $x=0$: $P=0$. $P^{\mathrm{plan}}_{1,1}=I_{1/6}(2,2)\approx0.042>0$.
    $\Delta f=\infty>\Omega=1$ (Cond.~1). $\EAC=\infty>1.05\times100$ (Cond.~2).
    Counter: $1\to0$.
  \item Termination executes at $t=2$ (the period after counter hits 0).
\end{itemize}

\paragraph{Settlement at $t_1^{\mathrm{term}}=2$, $P_{1,2}=0$.}
Outstanding advance recovery $= 0$ (no milestones certified).
\[
  R_1^{\mathrm{term}} = 0\times100 - (30-0) = -30.
\]

\paragraph{Cash flows.}
\begin{center}
\begin{tabular}{ccccc}
\toprule
$t$ & $x_{1,t}$ & Inflow & Net & Cumul.\\
\midrule
1 & 0 & $A_1=30$          & $+30$ & $+30$\\
2 & 0 & 0                 &   $0$ & $+30$\\
3 & 0 & $R^{\mathrm{term}}=-30$ & $-30$ & $0$\\
\bottomrule
\end{tabular}
\end{center}

\paragraph{Objective.}
\[
  Z^* = 30 + 0 - 30 = \mathbf{0}.
\]

\begin{notebox}
\NOTEH The advance payment is exactly recovered in the settlement.
No allocation is ever worthwhile; any spend at $t=1$ yields progress
too small to certify MS1 (given $\eta=0.05$ from $t=2$), and the settlement
simply grows by $P\times CP$ which is less than the spend. Hence $Z^*=0$
regardless of the allocation, and the MILP should allocate zero.
\end{notebox}

\paragraph{Assertions for pytest.}
\begin{enumerate}[label=A\arabic*.]
  \item $\sum_t x_{1,t} \leq 1.0$ (effectively zero spend).
  \item Project terminates at $t \leq 3$.
  \item $R_1^{\mathrm{term}} = -30 \pm 1$.
  \item No milestone certified.
  \item $|Z^* - 0| \leq 1.5$.
\end{enumerate}

\subsubsection{SP-3: Uniform Allocation, High Margin, Completion Optimal}
% ------------------------------------------------------------

\paragraph{Economic claim.}
With large profit margin and constant $\eta$, completing the project dominates
all alternatives (including termination). With $\gamma=1$, total net cash flow
is independent of timing within the feasible set, so any completing allocation
achieves $Z^*$.

\paragraph{Parameters.}
\[
  n=1,\quad s_1=1,\quad f_1=6,\quad \BAC_1=60,\quad \pi_1=0.50\Rightarrow CP_1=90.
\]
\[
  M_1=3,\quad \theta=(1/3,\;2/3,\;1),\quad \phi=(1/3,\;1/3,\;1/3),\quad
  \bar\eta=1.0.
\]
\[
  \alpha_1=0,\quad \rho_1=0,\quad
  \Omega_1=2,\quad \mu_1=0.50,\quad \tau_1^{\mathrm{tol}}=2.
\]

\paragraph{Milestone payments.}
\[
  R_{1,j}^{\mathrm{net}} = (1/3)\times90 = 30 \text{ per milestone.}
\]
Total inflow $= 90$. Total outflow (to complete) $= 60$.
\[
  Z^* = 90 - 60 = \mathbf{30}.
\]

\paragraph{Uniform allocation realisation.}
Spend $x_t = 10$ per period:
$P_{1,t} = 10t/60 = t/6$.
\begin{center}
\begin{tabular}{cccccc}
\toprule
$t$ & $x$ & $P_{1,t}$ & Inflow & Net & Cumul.\\
\midrule
1 & 10 & 1/6 &    0 & $-10$ & $-10$\\
2 & 10 & 2/6 & $30$ & $+20$ & $+10$\\
3 & 10 & 3/6 &    0 & $-10$ & $0$\\
4 & 10 & 4/6 & $30$ & $+20$ & $+20$\\
5 & 10 & 5/6 &    0 & $-10$ & $+10$\\
6 & 10 & 6/6 & $30$ & $+20$ & $+30$\\
\bottomrule
\end{tabular}
\end{center}
$Z = 30$. Any completing allocation gives the same $Z$ when $\gamma=1$.

\paragraph{Termination check.}
$\SPI \approx 1$ (actual tracks plan with uniform spend); $\CPI=1$. No conditions fire. \checkmark

\paragraph{Assertions for pytest.}
\begin{enumerate}[label=A\arabic*.]
  \item Total outflow $= 60 \pm 0.5$.
  \item All three milestones certified.
  \item Status: \texttt{completed}.
  \item $|Z^* - 30| \leq 1.0$.
  \item No termination conditions ever simultaneously active.
\end{enumerate}

\subsubsection{SP-4: Front-Loading Beats Uniform Allocation ($\gamma < 1$)}
% ------------------------------------------------------------

\paragraph{Economic claim.}
With discounting, earlier inflows are worth more. The MILP front-loads all
spending to certify milestones at $t=1$ and $t=2$, beating the uniform
allocation by a computable margin.

\paragraph{Parameters.}
\[
  n=1,\quad s_1=1,\quad f_1=8,\quad \BAC_1=100,\quad \pi_1=0.40\Rightarrow CP_1=140.
\]
\[
  M_1=2,\quad \theta=(0.5,\;1),\quad \phi=(0.5,\;0.5),\quad
  \bar\eta=1.0,\quad \gamma=0.95.
\]
\[
  \alpha_1=0,\quad \rho_1=0,\quad
  R_{1,j}^{\mathrm{net}}=70 \text{ each.}
\]

\paragraph{Front-loaded allocation.}
Spend $x_{1,1}=50$, $x_{1,2}=50$, rest zero:
\begin{align*}
  Z^{\mathrm{front}}
  &= \underbrace{\gamma^0 \times 70}_{\text{MS1 at }t=1}
   + \underbrace{\gamma^1 \times 70}_{\text{MS2 at }t=2}
   - \gamma^0\times50 - \gamma^1\times50\\
  &= 70 + 0.95\times70 - 50 - 0.95\times50\\
  &= 70 + 66.50 - 50 - 47.50 = \mathbf{39.00}.
\end{align*}

\paragraph{Uniform allocation baseline.}
Spend $x_t = 100/8 = 12.5$ per period. MS1 certified at $t=4$, MS2 at $t=8$.
\begin{align*}
  Z^{\mathrm{uniform}}
  &= \gamma^3\times70 + \gamma^7\times70
   - 12.5\sum_{t=0}^{7}\gamma^t\\
  &= 0.8574\times70 + 0.6983\times70
   - 12.5\times\frac{1-0.95^8}{0.05}\\
  &= 60.02 + 48.88 - 12.5\times6.423\\
  &= 108.90 - 80.29 = \mathbf{28.61}.
\end{align*}

Since $39.00 > 28.61$, front-loading is strictly better. The MILP must
choose front-loading.

\paragraph{Assertions for pytest.}
\begin{enumerate}[label=A\arabic*.]
  \item $x_{1,1} = 50\pm 0.5$, $x_{1,2} = 50\pm 0.5$.
  \item MS1 certified at $t=1$; MS2 certified at $t=2$.
  \item $Z^* \geq 38.5$.
  \item $Z^* > Z^{\mathrm{uniform}} = 28.61$.
\end{enumerate}

\paragraph{Economic claim.}
A project with zero $\eta$ from the start ($t \geq 1$) fires both termination
conditions at $t=1$. With $\tau^{\mathrm{tol}}=3$, the counter reaches zero at
$t=3$ and termination executes at $t=4$. The settlement exactly cancels the
advance.

\paragraph{Parameters.}
\[
  n=1,\quad s_1=1,\quad f_1=6,\quad \BAC_1=100,\quad CP_1=100,\quad \pi_1=0.
\]
\[
  M_1=1,\quad \theta=(1.0),\quad \phi=(1.0).

\[
  \alpha_1=0.20\Rightarrow A_1=20,\quad

\]
\[
  \tau_1^{\mathrm{tol}}=3,\quad \Omega_1=1,\quad \mu_1=0.10.
\]
\[
  \bar\eta_{1,t} = 0 \text{ for all } t.

\paragraph{Termination timeline.}
Since $\eta = 0$, no progress is possible regardless of spend. $P=0$ always.
$P^{\mathrm{plan}}>0$ from $t=1$: Condition~1 fires ($\Delta f=\infty$).

3 & 0 & 0        &   $0$ & $320$\\
5 & 0 & $-20$    & $-20$ & $320\to300$\\
\bottomrule
\end{tabular}
\end{center}

\paragraph{Objective.} $Z^* = 20 - 20 = \mathbf{0}$.

\paragraph{Assertions for pytest.}
\begin{enumerate}[label=A\arabic*.]

\end{enumerate}

\subsubsection{MP-2: Good Project vs.\ Bad Project (Unconstrained Cash)}
% ------------------------------------------------------------

\paragraph{Economic claim.}
With abundant cash, the MILP allocates zero to the bad project (which terminates
giving net zero) and maximises spend on the good project. $Z^* = Z_{\mathrm{Good}}$.

\paragraph{Parameters.}

\textit{Project Bad:}
$\BAC_B=100$, $CP_B=100$, $\pi_B=0$, $s_B=f_B^{\mathrm{plan}}=1\ldots8$,
$M=2$, $\theta=(0.5,1)$, $\alpha_B=0.30$ ($A_B=30$),
$\delta^{\mathrm{rec}}=0.25$, $\psi=0.10$, $\rho_B=0$,
$\Omega_B=1$, $\mu_B=0.10$, $\tau_B^{\mathrm{tol}}=2$.
\[
  \bar\eta_{B,1}=1.0,\quad \bar\eta_{B,t}=0.05 \text{ for } t\geq2.
\]

\textit{Project Good:}
$\BAC_G=80$, $CP_G=120$, $\pi_G=0.50$, $s_G=1$, $f_G=8$,
$M=2$, $\theta=(0.5,1)$, $\alpha_G=0$, $\rho_G=0$,
$\Omega_G=2$, $\mu_G=0.30$, $\tau_G^{\mathrm{tol}}=3$,
$\bar\eta_G=1.0$.

$H=8$. $B_1=3\times180=540$.

\paragraph{Analysis of Project Bad.}
From $t=2$, $\eta_B=0.05$. To reach MS1: need $50/0.05 = 1000 \gg \BAC$.
MS1 unreachable. With zero spend: counter decrements from $\tau^{\mathrm{tol}}=2$,
hitting zero at $t=3$, termination at $t=4$.
Settlement: $R_B^{\mathrm{term}} = 0 - 30 = -30$. Advance $= +30$.
Net contribution of Bad $= 30 - 30 = 0$.

\paragraph{Analysis of Project Good.}
Front-load: $x_{G,1}=40$, $x_{G,2}=40$.
$P_{G,1}=0.5$ (MS1, $R_{G,1}^{\mathrm{net}}=60$), $P_{G,2}=1.0$ (MS2, $R_{G,2}^{\mathrm{net}}=60$).
$Z_G = (60-40)+(60-40) = 40$.

\[
  Z^* = Z_B + Z_G = 0 + 40 = \mathbf{40}.
\]

\paragraph{Assertions for pytest.}
\begin{enumerate}[label=A\arabic*.]
  \item $\sum_t x_{B,t} \leq 2.0$.
  \item Project Bad terminates at $t \leq 4$.
  \item Project Good: \texttt{completed}.
  \item $|Z^* - 40| \leq 1.5$.
\end{enumerate}

\subsubsection{MP-4: Three Projects, Budget-Driven Termination of Low Project}
% ------------------------------------------------------------

\paragraph{Economic claim.}
Project~Low has $\eta=0.5$, so achieving MS1 requires spend that immediately
triggers EAC overrun. The MILP terminates Low and completes High and Mid,
achieving $Z^* = Z_H + Z_M + Z_L = 30 + 12 + 0 = 42$.

\paragraph{Parameters.}
$H=6$, $B_1=150$, $\gamma=1$. All: $\alpha=0$, $\rho=0$,
$\tau^{\mathrm{tol}}=2$, $\delta^{\mathrm{rec}}=0.25$, $\psi=0.10$.

\begin{center}
\begin{tabular}{lccccccc}
\toprule
Proj. & $\BAC$ & $\pi$ & $CP$ & $M$ & $\theta$ & $\bar\eta$ & $\Omega/\mu$\\
\midrule
High & 60 & 0.50 & 90 & 2 & $(0.5,1)$ & 1.0 & $2/0.40$\\
Mid  & 60 & 0.20 & 72 & 2 & $(0.5,1)$ & 1.0 & $2/0.30$\\
Low  & 60 & 0.10 & 66 & 2 & $(0.5,1)$ & 0.5 & $1/0.10$\\
\bottomrule
\end{tabular}
\end{center}

\paragraph{Why Low terminates.}
To reach MS1 ($\theta=0.5$) with $\eta=0.5$:
$x^{\mathrm{needed}} = 0.5 \times 60 / 0.5 = 60 = \BAC$.
After spending 60: $\ACWP=60$, $\EV=0.5\times60=30$, $\CPI=30/60=0.5$.
$\EAC = 60 + (60-30)/0.5 = 60+60=120 > (1+0.10)\times60 = 66$.
Condition~2 fires immediately. Even spending less does not help: any $x$ yields
$\CPI < 1$ from $t=1$, so $\EAC > 60$ always. Condition~2 always fires.
Condition~1 fires because progress is slower than plan with $\eta=0.5$.
\textbf{Low terminates after $\tau^{\mathrm{tol}}=2$ consecutive periods.}

\paragraph{Settlement for Low.}
$\alpha_L=0$, so $A_L=0$ and $R_L^{\mathrm{term}} = P_L \times CP_L \approx 0$
(minimal progress if zero spend). Net contribution $= 0$.

\paragraph{Optimal allocation (High and Mid).}
Spend 30 each at $t=1$ (certify MS1 of H and M simultaneously):
\begin{align*}
  P_{H,1}&=30/60=0.5 \Rightarrow R_{H,1}^{\mathrm{net}}=45,\quad
  P_{M,1}=30/60=0.5 \Rightarrow R_{M,1}^{\mathrm{net}}=36.
\end{align*}
$B_2 = 150 - 60 + (45+36) = 171$.
Spend 30 each at $t=2$ (certify MS2 of H and M):
\[
  R_{H,2}^{\mathrm{net}}=45,\quad R_{M,2}^{\mathrm{net}}=36.
\]
$B_3 = 171 - 60 + 81 = 192$.

$Z^* = (45-30)+(36-30)+(45-30)+(36-30) + 0 = 15+6+15+6 = \mathbf{42}$.

\paragraph{Assertions for pytest.}
\begin{enumerate}[label=A\arabic*.]
  \item High and Mid: \texttt{completed} by $t=2$.
  \item Low: \texttt{terminated} by $t \leq 4$; allocation to Low $\leq 2$.
  \item $|Z^* - 42| \leq 1.5$.
  \item $B_t \geq 0$ for all $t$.
\end{enumerate}

\section{Testing Architecture Notes}

\paragraph{Framework.} Use \texttt{pytest}. Install: \texttt{pip install pytest}.
All numerical comparisons use \texttt{pytest.approx(expected, abs=tol)}.

\paragraph{Fixture pattern.} Each case is a pytest fixture that builds a
hard-coded \texttt{CONFIG} dict, calls \texttt{generate\_portfolio},
\texttt{build\_and\_solve}, and \texttt{build\_records}, then exposes
\texttt{sol}, \texttt{rec}, \texttt{portfolio} to the test function.

\paragraph{Plot flag.} Add a \texttt{conftest.py} flag \texttt{--plots}; when
passed, each test saves the full per-project and portfolio plots to
\texttt{test\_plots/\{case\_id\}/}. This reuses the existing
\texttt{plot\_portfolio} function from \texttt{milp\_ppm\_test.py} directly.

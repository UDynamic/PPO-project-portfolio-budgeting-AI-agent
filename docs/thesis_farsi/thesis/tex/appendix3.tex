\chapter{نقشه‌یِ معماریِ مخزن و ماتریسِ ردیابیِ مفاهیم}
\label{app:repo_map}
%\thispagestyle{empty}

در این پیوست، ساختارِ مخزنِ کدِ همراهِ این پایان‌نامه — شاملِ
محیط، حل‌کننده‌هایِ معیار، مجموعه‌آزمونِ اعتبارسنجی، و خطِ لوله‌یِ
مرورِ ادبیاتِ توصیف‌شده در بخشِ مرورِ ادبیات — معرفی می‌شود و یک
ماتریسِ ردیابی ارائه می‌گردد که هر مفهوم و مشارکتِ مطرح‌شده در متنِ
اصلی را به محلِ متناظرِ آن در مخزن پیوند می‌دهد.

% ------------------------------------------------------------
\section{نمایِ کلیِ معماریِ مخزن}
\label{subsec:repo_overview}
% ------------------------------------------------------------

پیاده‌سازیِ کاملِ همراهِ این پایان‌نامه — محیط، حل‌کننده‌هایِ
معیار، مجموعه‌آزمونِ اعتبارسنجی، و خطِ لوله‌یِ مرورِ ادبیاتِ
توصیف‌شده در فصلِ دوم — به‌صورتِ یک مخزنِ واحدِ تحتِ کنترلِ نسخه
سازمان‌دهی شده است. این مخزن، \emph{مستندسازی} (این پایان‌نامه،
نسخه‌یِ انگلیسیِ آن، و پیکره‌یِ ادبیاتِ موضوع) را از
\emph{پیاده‌سازی} (محیط، دفترِ حسابرسی، حل‌کننده‌هایِ معیار، و
آزمون‌ها) جدا می‌کند، چنان‌که در زیر خلاصه شده است.

\begin{lstlisting}[basicstyle=\ttfamily\scriptsize, breaklines=true]
    ppm_env/
    |-- notation.yaml        فرهنگ‌واژه‌یِ پارامترها (نمادِ آکادمیک <-> متغیرِ کد)
    |-- docs/
    |   |-- article/          متنِ اصلیِ لاتکِ این نوشتار
    |   |-- lit_archive/      خطِ لوله‌یِ غربالگریِ ادبیات و پیکره‌یِ آن
    |   |-- mscn/             مستنداتِ تکمیلیِ پایان‌نامه
    |   |-- pp_presentation/  اسلایدهایِ دفاعِ پایان‌نامه (تحویلی دانشگاهی)
    |   `-- thesis_farsi/     نسخه‌یِ فارسیِ پایان‌نامه (تحویلی دانشگاهی)
    `-- src/
        |-- db/               طرحواره‌یِ دفترِ حسابرسی، پیکربندی‌ها، رکوردهایِ اجرا
        |-- env/               پیاده‌سازیِ هسته‌ایِ محیطِ CMDP
        `-- tests/             مجموعه‌آزمونِ اعتبارسنجی، معیارها، تحلیلِ شکاف
    \end{lstlisting}

پوشه‌هایِ \code{mscn/}، \code{pp\_presentation/}، و
\code{thesis\_farsi/}، الزاماتِ تکمیلِ مقطعِ دانشگاهی را برآورده
می‌کنند و خارج از دامنه‌یِ تکرارپذیریِ این پایان‌نامه قرار دارند؛
این‌ها فقط برایِ کامل‌بودنِ فهرست آورده شده‌اند. فایلِ
\code{notation.yaml} در ریشه‌یِ مخزن، تناظری صریح و
ماشین‌خوانا میانِ هر نمادِ آکادمیکِ معرفی‌شده در فصلِ سوم و
متغیرِ متناظرِ آن در کدِ منبع فراهم می‌کند و به‌عنوانِ مرجعِ
اصلیِ ارجاعِ متقابل برایِ خوانندگانی که می‌خواهند مدلِ رسمی را به
پیاده‌سازی پیوند دهند، در نظر گرفته شده است.

\medskip

خطِ لوله‌یِ مرورِ ادبیاتِ توصیف‌شده در بخشِ «پروتکل جستجو و
غربالگری» در \code{docs/lit\_archive/} قرار دارد و در کنارِ
اسکریپت‌هایِ پردازشی، بایگانیِ کاملِ \lr{PDF}‌ها
(\code{lit\_archive\_passed/}, \code{lit\_archive\_failed/}) و
رکوردهایِ استخراجِ ساخت‌یافته‌یِ \lr{JSON} را که جدولِ ترکیب و
نمودارهایِ تحلیلِ شکافِ گزارش‌شده در فصلِ دوم را تولید کرده‌اند،
نگه می‌دارد.

% - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
\section{ماژول‌هایِ اصلیِ مشارکت}
\label{subsec:repo_contrib}
% - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -

ماژول‌هایِ پیاده‌کننده‌یِ مدلِ رسمیِ فصلِ سوم و ارزیابیِ فصلِ
چهارم، به‌طورِ کامل در \code{src/} قرار دارند. دینامیکِ محیط —
ساختِ حالت، محاسبه‌یِ \lr{EVM}/منحنیِ \lr{S}، چرخه‌یِ جریانِ نقدیِ
\lr{FIDIC}، و دروازه‌یِ فسخ — در \code{src/env/} پیاده‌سازی شده‌اند،
به‌طوری‌که رابطِ عمومیِ سازگار با \lr{Gymnasium} و روال‌هایِ
پشتیبانِ داخلیِ محیط، به‌صورتِ ماژول‌هایِ جداگانه در همان پوشه
نگه‌داری می‌شوند. در حینِ آموزش یا ارزیابی، کلاسِ محیط به‌ازایِ
هر قسمت از طریقِ رابطِ استانداردِ \code{reset()}/\code{step()} در
\lr{Gymnasium} نمونه‌سازی می‌شود؛ هیچ مؤلفه‌یِ راه‌اندازِ اضافه‌ای
لازم نیست.

دفترِ حسابرسیِ \lr{SQLite} معرفی‌شده در فصلِ سوم، طرحواره‌بندی
شده و تحتِ \code{src/db/} پایدار می‌ماند؛ این پوشه همچنین
پیکربندی‌هایِ محکِ سبدِ پروژه‌یِ به‌کاررفته در فصلِ چهارم و
پایگاه‌دادهٔ رکوردهایِ اجرایِ تولیدشده توسطِ هر قسمتِ اجراشده را
نگه می‌دارد.

لایه‌یِ اعتبارسنجی و ارزیابی در \code{src/tests/} قرار دارد که
چهار مؤلفه‌یِ از نظرِ عملکردی متمایز را دربرمی‌گیرد: مجموعه‌آزمونِ
سازگاریِ \lr{Gymnasium}، حل‌کننده‌هایِ معیار و بهینه، تولیدکننده‌یِ
پیکربندیِ محک همراه با قسمت‌هایِ مرجعِ بایگانی‌شده‌یِ آن، و
اسکریپت‌هایِ تحلیلِ پسینی که برایِ تولیدِ شکل‌ها و جدول‌هایِ فصلِ
چهارم به‌کار رفته‌اند.

\begin{lstlisting}[basicstyle=\ttfamily\scriptsize, breaklines=true]
    src/tests/
    |-- gym_test.py                 مجموعه‌آزمونِ سازگاری با Gymnasium API
    |-- gym_certificate.png         گواهیِ سازگاری (پیوست ب)
    |-- baseline_solvers.py         اکتشافی‌هایِ حریصانه و حل‌کننده‌یِ بهینه
    |-- reference_episodes_gen/     تولیدکننده‌یِ پیکربندیِ محک
    |-- reference_episodes_record/  پیکربندی‌ها و جواب‌هایِ بایگانی‌شده
    `-- analysis/                   اسکریپت‌هایِ تحلیلِ شکاف و تولیدِ شکل
    \end{lstlisting}

پوشه‌یِ \code{reference\_episodes\_record/} مجموعه‌یِ پیکربندی و
جواب‌هایِ ثبت‌شده‌یِ به‌کاررفته برایِ اعتبارسنجیِ محیط و تولیدِ چهار
پیکربندیِ محکِ ارزیابی‌شده در فصلِ چهارم را نگه می‌دارد؛ این پوشه
به‌عنوانِ یک الگویِ کاری برایِ کسانی در نظر گرفته شده که می‌خواهند
محیط را بر اساسِ داده‌هایِ شرکتِ خود نمونه‌سازی کنند.

\section{یادداشت‌هایِ تکرارپذیری}

% - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -

نگاشتِ ارائه‌شده در جدولِ \ref{tab:repo_map} بازتاب‌دهنده‌یِ یک
تصویرِ لحظه‌ایِ مشخص از مخزن در زمانِ نگارش است؛ توسعه‌یِ بعدی
ممکن است ماژول‌هایِ منفرد را بدونِ تأثیر بر ساختارِ سطحِ پوشه‌یِ
توصیف‌شده در این پیوست، بازسازماندهی کند. دسترسی به مخزن —
شاملِ تاریخچه‌یِ کاملِ \lr{commit}‌ها، پیکربندی‌هایِ محک، و
گواهیِ سازگاریِ \lr{Gymnasium} بازتولیدشده در پیوستِ ب — از طریقِ
نویسنده‌یِ مسئول و بنا به درخواست در دسترس است.

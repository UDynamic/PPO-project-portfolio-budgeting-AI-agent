% ── Schedule ──────────────────────────────────────────────────────────────────
\rowcolor{headbg}
\multicolumn{4}{l}{\textcolor{headfg}{\textit{Schedule}}} \\

Planned start
  & $s_i$ & $\mathbb{Z}_{\geq 0}$
  & Scheduled episode period at which project $i$ begins
    (\code{planned\_start}). \\
\rowcolor{RowAlt}
Planned duration
  & $D_i$ & $\mathbb{Z}_{\geq 0}$
  & Number of active periods planned for project $i$
    (\code{planned\_duration}); $D_i=0$ collapses the S-curve
    to a step function \eqref{eq:scurve}. \\
Planned finish
  & $f_i$ & $\mathbb{Z}_{\geq 0}$
  & $f_i = s_i + D_i$; derived at reset; defines the hard
    deadline via \eqref{eq:horizon} and \eqref{eq:term_cond2}. \\
\rowcolor{RowAlt}
S-curve shape
  & $a_i,\; b_i$ & $\mathbb{R}^+$
  & Beta CDF shape parameters governing $PV_i(\tau)$
    \eqref{eq:scurve} (\code{scurve\_a}, \code{scurve\_b}). \\
Milestone progress threshold
  & $\psi_{i,j}$ & $(0,1]$
  & Minimum $EV_{i,t}$ required to certify milestone $j$
    (\code{progress\_threshold}). \\
\rowcolor{RowAlt}
Milestone timestep threshold
  & $\tau^{\mathrm{ms}}_{i,j}$ & $\mathbb{Z}_{\geq 0}$
  & Earliest project-time for milestone $j$ certification;
    episode threshold $= s_i + \tau^{\mathrm{ms}}_{i,j}$
    \citep{FIDIC2017} (\code{timestep\_threshold}). \\
Progress delay cap
  & $\delta^{\max}_i$ & $[0,1]$
  & Maximum admissible lag before breach flag
    $\mathbb{1}^{\delta}_{i,t}$ activates \eqref{eq:term_cond1}
    (\code{progress\_delay\_cap}). \\
\rowcolor{RowAlt}
Finish delay cap
  & $\Delta^{\max}_i$ & $\mathbb{Z}_{>0}$
  & Extra periods beyond $f_i$ before the hard deadline fires
    \eqref{eq:term_cond2} (\code{finish\_delay\_cap}). \\
Termination tolerance
  & $\theta_i$ & $\mathbb{Z}_{>0}$
  & Grace-period length; models FIDIC Sub-Clause~15.1
    \citep{FIDIC2017, Ndekugri2008}
    (\code{termination\_tolerance}). \\

% ── Contract ──────────────────────────────────────────────────────────────────
\rowcolor{headbg}
\multicolumn{4}{l}{\textcolor{headfg}{\textit{Contract}}} \\

\rowcolor{RowAlt}
Budget at completion
  & $\mathrm{BAC}_i$ & $\mathbb{R}_{>0}$
  & Total planned contractor cost (\code{bac}). \\
Profit margin
  & $\pi_i$ & $\mathbb{R}_{\geq 0}$
  & $P_i = \mathrm{BAC}_i(1+\pi_i)$
    (\code{profit\_percent}). \\
\rowcolor{RowAlt}
Advance payment rate
  & $\phi_i$ & $[0,0.3]$
  & $\mathrm{Adv}_i = \phi_i P_i$ disbursed at $t = s_i$;
    $\phi_i = 0$ disables the advance mechanism
    (\code{advance\_percent}). \\
Advance recovery threshold
  & $\psi^{\mathrm{rec}}_i$ & $[0,1]$
  & Cumulative certified gross fraction above which recovery
    deductions commence \eqref{eq:advance_rec}
    (\code{advance\_trigger}). \\
\rowcolor{RowAlt}
Advance recovery rate
  & $\delta^{\mathrm{rec}}_i$ & $[0,1]$
  & Per-milestone deduction rate applied to recover
    $\mathrm{Adv}_i$ \eqref{eq:advance_rec}
    (\code{advance\_recovery}). \\
Retention rate
  & $\nu_i$ & $[0,0.1]$
  & $R^{\mathrm{held}}_{i,j} = \nu_i G_{i,j}$; released in
    full at $j = M_i$ (\code{retention\_rate}). \\
\rowcolor{RowAlt}
Milestone weight
  & $w_{i,j}$ & $[0,1]$
  & Fraction of $P_i$ paid at milestone $j$;
    $\sum_j w_{i,j} = 1$ (\code{payment\_weight}). \\

% ── Portfolio ─────────────────────────────────────────────────────────────────
\rowcolor{headbg}
\multicolumn{4}{l}{\textcolor{headfg}{\textit{Portfolio}}} \\

Initial budget
  & $B_0$ & $\mathbb{R}_{>0}$
  & $B_0 = \kappa \sum_i \mathrm{BAC}_i$; derived at reset. \\
\rowcolor{RowAlt}
Budget tightness
  & $\kappa$ & $\mathbb{R}_{>0}$
  & Liquidity regime scalar (\code{budget\_tightness}). \\
Discount factor
  & $\gamma$ & $(0,1]$
  & Per-period discount applied to net cash-flow rewards
    \citep{Damodaran2012} (\code{discount}). \\
\rowcolor{RowAlt}
Annual interest rate
  & $r$ & $[0,1]$
  & Monthly treasury-draw cost $= r/12$ per
    \eqref{eq:interest}; plain scalar, not a distribution entry
    (\code{annual\_interest\_rate}). \\
Episode horizon
  & $T$ & $\mathbb{Z}_{>0}$
  & Derived from \eqref{eq:horizon}; not a config entry. \\

% ── Stochastic Primitive ──────────────────────────────────────────────────────
\rowcolor{headbg}
\multicolumn{4}{l}{\textcolor{headfg}{\textit{Stochastic Primitive}}} \\

\rowcolor{RowAlt}
Work efficiency
  & $\eta_{i,t}$ & ${\geq}0.01$
  & Sole stochastic primitive; drawn per period per project
    from a configurable distribution; \code{custom\_ordinal}
    cycles deterministically for reproducible benchmarking;
    realised after $x_{i,t}$ is committed
    (\code{efficiency}). \\

% ── Structural ────────────────────────────────────────────────────────────────
\rowcolor{headbg}
\multicolumn{4}{l}{\textcolor{headfg}{\textit{Structural (implementation)}}} \\

Portfolio size
  & $N$ & $\mathbb{Z}_{>0}$
  & Number of projects; sampled at each \code{reset()} if
    configured as a distribution; determines observation and
    action space dimensions (\code{n\_projects}). \\
\rowcolor{RowAlt}
Milestone count
  & $M_i$ & $\mathbb{Z}_{>0}$
  & Number of milestones per project $i$; determines the
    lengths of $\{\psi_{i,j}\}$, $\{w_{i,j}\}$, and
    $\{\tau^{\mathrm{ms}}_{i,j}\}$
    (\code{n\_milestones}). \\

\renewcommand{\arraystretch}{1.3}
\setlength{\tabcolsep}{5pt}
\begin{tabular}{p{3.4cm} p{2.0cm} p{1.2cm} p{6.8cm}}
  \toprule
  \textbf{Parameter} & \textbf{Symbol} &
  \textbf{Domain} & \textbf{Definition} \\
  \midrule
  \multicolumn{4}{l}{\textit{Schedule}} \\
  \midrule
  Planned start / finish
    & $s_i,\; f_i$ & $\mathbb{Z}_{\geq 0}$
    & $f_i = s_i + D_i$; anchor of S-curve and activity window. \\
  \rowcolor{RowAlt}
  Planned duration
    & $D_i$ & $\mathbb{Z}_{>0}$
    & $D_i = f_i - s_i$; normalises project-time for S-curve. \\
  S-curve shape
    & $a_i,\; b_i$ & $\mathbb{R}^+$
    & Beta CDF parameters governing $PV_i(\tau)$ \eqref{eq:scurve}. \\
  \rowcolor{RowAlt}
  Milestone progress threshold
    & $\psi_{i,j}$ & $(0,1]$
    & Minimum $EV_{i,t}$ to certify milestone $j$;
      $\psi_{i,M_i}=1$. \\
  Milestone timestep threshold
    & $\tau^{\mathrm{ms}}_{i,j}$ & $\mathbb{Z}_{\geq 0}$
    & Earliest project-time for certification;
      episode threshold $= s_i + \tau^{\mathrm{ms}}_{i,j}$
      \citep{FIDIC2017}. \\
  \rowcolor{RowAlt}
  Progress delay cap
    & $\delta^{\max}_i$ & $[0,1]$
    & Maximum lag $\delta_{i,t}$ before
      $\mathbb{1}^{\delta}_{i,t}$ activates. \\
  Finish delay cap
    & $\Delta^{\max}_i$ & $\mathbb{Z}_{>0}$
    & Extra periods after $f_i$ before hard deadline fires. \\
  \rowcolor{RowAlt}
  Termination tolerance
    & $\theta_i$ & $\mathbb{Z}_{>0}$
    & Grace-period length; models FIDIC Sub-Clause~15.1
      \citep{FIDIC2017, Ndekugri2008}. \\
  \midrule
  \multicolumn{4}{l}{\textit{Contract}} \\
  \midrule
  Budget at completion
    & $\mathrm{BAC}_i$ & $\mathbb{R}_{>0}$
    & Total planned contractor cost; EVM normalisation basis. \\
  \rowcolor{RowAlt}
  Profit margin
    & $\pi_i$ & $\mathbb{R}_{\geq 0}$
    & Mark-up rate; $P_i = \mathrm{BAC}_i(1+\pi_i)$. \\
  Advance payment rate
    & $\phi_i$ & $[0,0.3]$
    & $\mathrm{Adv}_i = \phi_i P_i$ received at $t=s_i$. \\
  \rowcolor{RowAlt}
  Advance recovery rate
    & $\rho_i$ & $[0,1]$
    & Fraction of each milestone gross payment withheld for
      advance recovery:
      $A^{\mathrm{rec}}_{i,j} = \min(\rho_i G_{i,j},
      \text{advance remaining})$. \\
  Retention rate
    & $\nu_i$ & $[0,0.1]$
    & Fraction of each gross payment withheld;
      $R^{\mathrm{held}}_{i,j} = \nu_i G_{i,j}$;
      released in full at $j = M_i$. \\
  \rowcolor{RowAlt}
  Milestone weight
    & $w_{i,j}$ & $[0,1]$
    & Fraction of $P_i$ allocated to milestone $j$;
      $\sum_j w_{i,j}=1$. \\
  \midrule
  \multicolumn{4}{l}{\textit{Portfolio}} \\
  \midrule
  Initial budget
    & $B_0$ & $\mathbb{R}_{>0}$
    & $B_0 = \kappa \sum_i \mathrm{BAC}_i$. \\
  \rowcolor{RowAlt}
  Budget tightness
    & $\kappa$ & $\mathbb{R}_{>0}$
    & Liquidity regime; $\kappa \gg 1$: relaxed;
      $\kappa \approx 1$: tight; $\kappa \ll 1$: scarce. \\
  Discount factor
    & $\gamma$ & $(0,1]$
    & Per-period discount; shared across all projects
      \citep{Damodaran2012}. \\
  \rowcolor{RowAlt}
  Annual interest rate
    & $r$ & $[0,1]$
    & Monthly treasury-draw interest $= r/12$
      per \eqref{eq:interest}. \\
  Episode horizon
    & $T$ & $\mathbb{Z}_{>0}$
    & Derived from \eqref{eq:horizon}; not a fixed input. \\
  \midrule
  \multicolumn{4}{l}{\textit{Stochastic Primitive}} \\
  \midrule
  \rowcolor{RowAlt}
  Work efficiency
    & $\eta_{i,t}$ & $\geq 0.01$
    & Sole stochastic primitive; drawn per period per project
      from a configurable distribution (fixed, uniform, Beta,
      triangular, lognormal, or custom ordinal sequence);
      realised after $x_{i,t}$ is committed. \\
  \bottomrule
\end{tabular}

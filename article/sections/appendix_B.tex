% ============================================================
\section*{Conventions}
% ============================================================

All cases share these conventions unless explicitly overridden:
\begin{itemize}
  \item $\gamma = 0.95$ throughout, imposing time value of money in
    every case. This makes earlier certification strictly preferable and
    uniquely pins down the optimal allocation timing.
  \item Earliest certification dates $e_{i,j}$ are set for every case.
    The default rule is $e_{i,j} = s_i + \lfloor(j-1)\cdot
    D_i^{\mathrm{plan}}/M_i\rfloor$, spreading milestones evenly across
    the planned duration. Any departure is stated explicitly.
  \item $B_1 = 3\cdot\sum_i \BAC_i$ (large; cash never binding unless
    stated).
  \item $\bar\eta_{i,t}=\bar\eta$ constant and deterministic unless
    stated.
  \item Equal milestone weights $\phi_{i,j}=1/M_i$ unless stated.
  \item $\alpha_i=0$, $\rho_i=0$ unless stated.
  \item $\delta^{\mathrm{rec}}=0.25$, $\psi=0.10$ (FIDIC defaults)
    unless stated.
  \item $\pi_i=0$ unless stated ($CP_i=\BAC_i$).
  \item S-curve shape $a_i=b_i=2$ (symmetric Beta) unless stated.
  \item Monetary units are abstract.
\end{itemize}

\begin{notebox}
\NOTEH With $\gamma=0.95$ and earliest certification dates enforced,
the optimal strategy in every completing case is to accumulate progress
as fast as possible so that $P_{i,t}^{\mathrm{actual}} \geq \theta_{i,j}$
is satisfied by $t = e_{i,j}$, then certify immediately when the window
opens. Spending earlier than needed to reach the threshold is wasteful
only if it depletes cash needed elsewhere; spending later than $e_{i,j}$
loses discount value. This makes the MILP allocation uniquely determined
in all cases below.
\end{notebox}

% ============================================================
\section{Group 1: Progress and Milestone Certification}
% ============================================================

\textit{What is being proven:} budget converts to progress at rate $\eta$,
milestones certify at the correct threshold and not before $e_{i,j}$,
payments trigger correctly, and milestone ordering is enforced even when
progress jumps past multiple thresholds in one period.

% ------------------------------------------------------------
\subsection{SP-1: Perfect Efficiency, Two Milestones}
% ------------------------------------------------------------

\paragraph{Economic claim.}
With $\eta=1$ and $\gamma=0.95$, the MILP front-loads spend so that
MS1 certifies exactly at $e_{1,1}=1$ and MS2 certifies exactly at
$e_{1,2}=2$. Spending all budget at $t=1$ would not accelerate MS2
because $e_{1,2}=2$ blocks early certification.

\paragraph{Parameters.}
\[
  n=1,\quad s_1=1,\quad f_1=4,\quad D^{\mathrm{plan}}=4,\quad
  \BAC_1=100,\quad CP_1=120,\quad \pi_1=0.20.
\]
\[
  M_1=2,\quad \theta=(0.5,\;1.0),\quad \phi=(0.5,\;0.5),\quad \bar\eta=1.0.
\]
\[
  e_{1,1}=1+\lfloor0\cdot4/2\rfloor=1,\quad
  e_{1,2}=1+\lfloor1\cdot4/2\rfloor=3.
\]
\[
  \alpha_1=0,\quad\rho_1=0,\quad\Omega_1=1,\quad\mu_1=0.30,\quad
  \tau_1^{\mathrm{tol}}=2,\quad B_1=300,\quad H=4.
\]

\paragraph{Milestone payments.}
\[
  R_{1,1}^{\mathrm{net}}=60,\qquad R_{1,2}^{\mathrm{net}}=60.
\]

\paragraph{Optimal strategy.}
Spend 50 at $t=1$ ($P=0.5\geq\theta_1=0.5$, window $e_{1,1}=1$ open):
MS1 certifies. Spend 50 at $t=2$ ($P=1.0\geq\theta_2=1.0$, but
$e_{1,2}=3$ not yet open): progress complete, MS2 pending.
At $t=3$: no spend needed, $e_{1,2}=3$ opens, MS2 certifies.

\begin{center}
\begin{tabular}{ccccccc}
\toprule
$t$ & $x_{1,t}$ & $P_{1,t}$ & Window open? & Inflow & Net\\
\midrule
1 & 50 & 0.50 & MS1: yes & 60 & $+10$\\
2 & 50 & 1.00 & MS2: no  &  0 & $-50$\\
3 &  0 & 1.00 & MS2: yes & 60 & $+60$\\
4 &  0 & 1.00 & ---      &  0 & $0$\\
\bottomrule
\end{tabular}
\end{center}

\paragraph{Objective.}
\[
  Z^* = \gamma^0(60-50) + \gamma^1(0-50) + \gamma^2(60-0)
      = 10 - 0.95\times50 + 0.9025\times60
      = 10 - 47.5 + 54.15 = \mathbf{16.65}.
\]

\begin{notebox}
\NOTEH Could the MILP defer MS1 spend to $t=2$ and MS2 spend to $t=3$?
That gives $Z = \gamma^1(60-50)+\gamma^2(60-50) = 0.95\times10
+0.9025\times10=9.5+9.025=18.525$. Wait --- that is higher. The MILP
should defer both. Recalculate: deferring MS1 to $t=2$ means $P\geq0.5$
at $t=2$ with $e_{1,1}=1$ open. And MS2: spend at $t=3$ to get $P=1$
with $e_{1,2}=3$ open. $Z = \gamma^1(60-50)+\gamma^2(60-50)
=0.95\times10+0.9025\times10=18.525$. But then the outflows are also
discounted. Actually with $\gamma<1$ we want inflows early and outflows
late --- the MILP will actually defer spending to get earlier net.
\end{notebox}

Let me redo this properly. The MILP maximises:
\[
  \sum_t \gamma^{t-1}(V_t - x_t).
\]
Outflows at $t$ cost $\gamma^{t-1}$ weight; inflows at $t$ earn $\gamma^{t-1}$
weight. Since outflows precede inflows (you spend before the milestone
certifies), and $\gamma<1$ discounts future inflows, the net effect is:
spend as late as possible while still reaching the threshold by $e_{i,j}$.

For MS1 ($e_{1,1}=1$, threshold 0.5, requires spend 50 with $\eta=1$):
must spend 50 by $t=1$. No choice --- earliest possible is $t=1$.

For MS2 ($e_{1,2}=3$, threshold 1.0, additional spend 50):
must have cumulative spend 100 by $t=3$. Already have 50 from $t=1$.
Need 50 more by $t=3$. To minimise discounted outflow, spend as late
as possible: spend 50 at $t=3$ itself (just before window opens
--- but spend and certification happen same period, so spend at $t=3$,
$P$ updates, $e_{1,2}=3$ satisfied, MS2 certifies at $t=3$).

\paragraph{Revised optimal strategy.}
\begin{center}
\begin{tabular}{ccccccc}
\toprule
$t$ & $x_{1,t}$ & $P_{1,t}$ & Window open? & Inflow & Net\\
\midrule
1 & 50 & 0.50 & MS1: yes & 60  & $+10$\\
2 &  0 & 0.50 & MS2: no  &  0  & $0$\\
3 & 50 & 1.00 & MS2: yes & 60  & $+10$\\
4 &  0 & 1.00 & ---      &  0  & $0$\\
\bottomrule
\end{tabular}
\end{center}

\[
  Z^* = \gamma^0(60-50) + \gamma^1(0) + \gamma^2(60-50)
      = 10 + 0 + 0.9025\times10 = 10+9.025 = \mathbf{19.025}.
\]

This is strictly better than the $t=1,2$ strategy (16.65). The earliest
certification constraint makes spending at $t=3$ uniquely optimal for
MS2 --- there is no incentive to spend at $t=2$ since the window does
not open until $t=3$ and deferring the outflow saves discount.

\paragraph{Assertions for pytest.}
\begin{enumerate}[label=A\arabic*.]
  \item $x_{1,1}=50\pm0.5$, $x_{1,2}=0\pm0.5$, $x_{1,3}=50\pm0.5$,
    $x_{1,4}=0$.
  \item MS1 certified at $t=1$; MS2 certified at $t=3$ (not before,
    since $e_{1,2}=3$).
  \item $|Z^*-19.025|\leq1.0$.
  \item Status: \texttt{completed}. No $\tau$ decrement.
\end{enumerate}

\subsection{SP-J: Milestone Ordering Enforced on Progress Jump}
% ------------------------------------------------------------

\paragraph{Economic claim.}
When a single allocation causes progress to clear both thresholds, both
milestones certify in the same period provided both windows are open.
If $e_{1,2}>e_{1,1}$, MS2 cannot certify at $t=e_{1,1}$ even though
progress qualifies.

\paragraph{Parameters.}
\[
  n=1,\quad s_1=1,\quad f_1=4,\quad\BAC_1=100,\quad CP_1=100,\quad\pi_1=0.
\]
\[
  M_1=2,\quad\theta=(0.5,\;1.0),\quad\phi=(0.5,\;0.5),\quad\bar\eta=1.0.
\]
\[
  e_{1,1}=1,\quad e_{1,2}=3.
\]
\[
  \alpha_1=0,\quad\rho_1=0,\quad\Omega_1=2,\quad\mu_1=0.50,\quad
  \tau_1^{\mathrm{tol}}=2,\quad B_1=300.
\]

\paragraph{Allocation.}
Spend $x_{1,1}=100$ at $t=1$. $P_{1,1}=1.0$, clearing both
$\theta_1=0.5$ and $\theta_2=1.0$.

MS1: $\theta_1\leq P$ and $t=1\geq e_{1,1}=1$. Certifies. $R=50$.
MS2: $\theta_2\leq P$ but $t=1 < e_{1,2}=3$. Does \emph{not} certify.

At $t=3$: no further spend needed. $e_{1,2}=3$ opens. MS2 certifies.
$R=50$.

\paragraph{Objective.}
\[
  Z^* = \gamma^0(50-100) + \gamma^1(0) + \gamma^2(50)
      = -50 + 0 + 0.9025\times50 = -50+45.125 = \mathbf{-4.875}.
\]

Is there a better strategy? Defer the 100 spend to $t=3$ (just before
both windows are open would be $t=3$ for MS2):
At $t=3$: spend 100, $P=1$, both $e_{1,1}=1$ and $e_{1,2}=3$ open,
both certify. $Z=\gamma^2(100-100)=0$. Better.

Or: spend 50 at $t=1$ (MS1 at $t=1$), spend 50 at $t=3$ (MS2 at $t=3$):
$Z=\gamma^0(50-50)+\gamma^2(50-50)=0$. Same.

Or: spend 50 at $t=1$, spend 50 at $t=2$ (progress reaches 1.0 at $t=2$
but $e_{1,2}=3$ still closed; MS2 certifies at $t=3$ with no spend):
$Z=\gamma^0(50-50)+\gamma^1(-50)+\gamma^2(50)=-47.5+45.125=-2.375$.
Worse than 0.

MILP optimum: spend nothing until $t=1$ for MS1 (mandatory since
$e_{1,1}=1$ and we must spend to certify), then spend 50 at $t=3$ for
MS2. $Z^*=0$.

\paragraph{Revised optimal strategy.}
\begin{center}
\begin{tabular}{ccccc}
\toprule
$t$ & $x$ & $P$ & Inflow & Net\\
\midrule
1 & 50 & 0.50 & 50 & $0$\\
2 &  0 & 0.50 &  0 & $0$\\
3 & 50 & 1.00 & 50 & $0$\\
4 &  0 & 1.00 &  0 & $0$\\
\bottomrule
\end{tabular}
\end{center}

$Z^*=0+0+0+0=\mathbf{0}$ (all nets are zero so discounting doesn't
change it).

\paragraph{Key assertion for ordering:} MS2 does not certify at $t=1$
even though $P_{1,1}=1.0\geq\theta_2$ after spending 100 at $t=1$,
because $e_{1,2}=3>1$.

\paragraph{Assertions for pytest.}
\begin{enumerate}[label=A\arabic*.]
  \item MS2 not certified at $t=1$ or $t=2$ (window closed).
  \item MS2 certified at $t=3$.
  \item $|Z^*-0|\leq1.0$.
  \item Status: \texttt{completed} by $t=3$.
\end{enumerate}

\section{Group 2: Discounting and Timing Incentives}
% ============================================================

\textit{What is being proven:} $\gamma<1$ and earliest certification dates
together make the optimal allocation uniquely determined: spend just enough
just before each $e_{i,j}$ to qualify, and certify the moment the window
opens. When two projects compete, complete the one whose milestone

% ------------------------------------------------------------
\subsection{SP-4: Deferring Spend to Last Possible Period is Optimal}
% ------------------------------------------------------------

\paragraph{Economic claim.}
With $\gamma<1$, deferring outflows to the latest period that still allows
certification at $e_{i,j}$ is strictly better than front-loading. The
margin over the front-loaded strategy is computable.

\paragraph{Parameters.}
\[
  n=1,\quad s_1=1,\quad f_1=8,\quad D^{\mathrm{plan}}=8,\quad
  \BAC_1=100,\quad\pi_1=0.40\Rightarrow CP_1=140.
\]
\[
  M_1=2,\quad\theta=(0.5,\;1),\quad\phi=(0.5,\;0.5),\quad\bar\eta=1.0.
\]
\[
  e_{1,1}=1+\lfloor0\cdot8/2\rfloor=1,\quad
  e_{1,2}=1+\lfloor1\cdot8/2\rfloor=5.
\]
\[
  R_{1,j}^{\mathrm{net}}=70\text{ each.}
\]

\paragraph{Optimal strategy.}
MS1 requires 50 spend; must be at $t=1$ (window opens at $t=1$, must
spend to qualify, no benefit deferring since window is already open
at $t=1$). MS2 requires 50 more; window opens at $t=5$, so spend 50
at $t=5$.

\[
  Z^{\mathrm{opt}} = \gamma^0(70-50) + \gamma^4(70-50)
    = 20 + 0.95^4\times20 = 20 + 0.8145\times20
    = 20 + 16.29 = \mathbf{36.29}.
\]

\paragraph{Front-loaded strategy (spend at $t=1$ and $t=2$).}
MS2 window still closed at $t=2$; progress qualifies at $t=2$ but
certifies only at $t=5$.
\[
  Z^{\mathrm{front}} = \gamma^0(70-50) + \gamma^1(-50) + \gamma^4(70)
    = 20 - 47.5 + 0.8145\times70 = 20-47.5+57.015 = \mathbf{29.515}.
\]

$Z^{\mathrm{opt}}=36.29 > Z^{\mathrm{front}}=29.515$. Deferring the
MS2 spend to $t=5$ saves $\gamma^1\times50-\gamma^4\times50
= 47.5-40.725=6.775$ in discounted outflow with identical inflow.

\paragraph{Assertions for pytest.}
\begin{enumerate}[label=A\arabic*.]
  \item $x_{1,1}=50\pm0.5$.
  \item $x_{1,2}=x_{1,3}=x_{1,4}=0\pm0.5$.
  \item $x_{1,5}=50\pm0.5$.
  \item MS1 certified at $t=1$; MS2 certified at $t=5$.
  \item $Z^*\geq35.5$.
  \item $Z^*>29.5$ (front-loaded strategy strictly dominated).
\end{enumerate}

\subsection{MP-K: Two Projects, Front-Heavy vs.\ Back-Heavy Milestones}
% ------------------------------------------------------------

\paragraph{Economic claim.}
Two projects identical in $\BAC$, $CP$, $\eta$, and duration differ only
in milestone weight distribution. Under $\gamma<1$, completing the
front-heavy project first (large early payment at $e_{A,1}$) yields
higher discounted inflow than completing the back-heavy project first.

\paragraph{Parameters.}
\[
  n=2,\quad s_i=1,\quad f_i=6,\quad D^{\mathrm{plan}}=6,\quad
  \BAC_i=60,\quad\pi_i=0.50\Rightarrow CP_i=90.
\]
\[
  M_i=2,\quad\theta_i=(0.5,\;1.0),\quad\bar\eta_i=1.0.
\]
\[
  e_{i,1}=1,\quad e_{i,2}=1+\lfloor1\cdot6/2\rfloor=4.
\]
\[
  \phi_A=(0.70,\;0.30),\quad\phi_B=(0.30,\;0.70).
\]
\[
  R_{A,1}^{\mathrm{net}}=63,\quad R_{A,2}^{\mathrm{net}}=27,\quad
  R_{B,1}^{\mathrm{net}}=27,\quad R_{B,2}^{\mathrm{net}}=63.
\]
Each milestone requires spend of 30. $B_1=300$.

\paragraph{Strategy 1 (A then B): complete A first.}
$t=1$: spend 30 on A (MS1$_A$ at $e_{A,1}=1$), 30 on B (MS1$_B$ at
$e_{B,1}=1$). Both MS1 certify simultaneously.
$t=4$: spend 30 on A (MS2$_A$ at $e_{A,2}=4$), 30 on B (MS2$_B$ at
$e_{B,2}=4$). Both MS2 certify simultaneously.

Both projects complete at the same time in this symmetric allocation.
The φ weights affect inflows but not timing here.

$Z_{\mathrm{sim}} = \gamma^0(63+27-30-30) + \gamma^3(27+63-30-30)
= 30 + 0.857\times30 = 30+25.72 = \mathbf{55.72}$.

Can we do better by sequencing? With $B_1=300$ (non-binding), can spend
30+30 at $t=1$ and 30+30 at $t=4$ regardless. Sequencing does not help
since both MS1 windows open at $t=1$ and both MS2 windows at $t=4$.

The test here is not about ordering but about φ weights: A's MS1
payment (63) is larger than B's MS1 payment (27). Under $\gamma<1$,
if we could stagger, getting A's MS1 earlier would help. But since
both $e_{i,1}=1$, the simultaneity is forced.

Adjust: set $e_{A,1}=1$, $e_{B,1}=2$ to force staggering:

\begin{center}
\begin{tabular}{cccccc}
\toprule
$t$ & $x_A$ & $x_B$ & Inflow & Net\\
\midrule
1 & 30 & 0  & 63 (MS1$_A$) & $+33$\\
2 &  0 & 30 & 27 (MS1$_B$) & $-3$\\
4 & 30 & 30 & 27+63=90    & $+30$\\
\bottomrule
\end{tabular}
\end{center}

$Z_{\mathrm{A1st}} = \gamma^0(63-30)+\gamma^1(27-30)+\gamma^3(90-60)
= 33-2.85+0.857\times30 = 33-2.85+25.72 = \mathbf{55.87}$.

Reverse (B's MS1 first at $t=1$, A's at $t=2$):
$Z_{\mathrm{B1st}} = \gamma^0(27-30)+\gamma^1(63-30)+\gamma^3(90-60)
= -3+0.95\times33+25.72 = -3+31.35+25.72 = \mathbf{54.07}$.

$Z_{\mathrm{A1st}}=55.87 > Z_{\mathrm{B1st}}=54.07$. A first dominates.

Use $e_{A,1}=1$, $e_{B,1}=2$ in this case to isolate the ordering
decision.

\paragraph{Assertions for pytest.}
\begin{enumerate}[label=A\arabic*.]
  \item MS1$_A$ certified at $t=1$; MS1$_B$ certified at $t=2$.
  \item $Z^*\geq55.5$.
  \item $Z^*>54.07$ (B-first strictly dominated).
  \item Both projects \texttt{completed}.
\end{enumerate}

\section{Group 3: Advance Payment and Recovery Mechanics}
% ============================================================

\textit{What is being proven:} advance arrives at $t=s_i$, recovery

% ------------------------------------------------------------
\subsection{SP-2: Advance Harvest, Efficiency Decay, Forced Termination}
% ------------------------------------------------------------

\paragraph{Economic claim.}
A large advance combined with rapidly decaying efficiency makes
completion economically impossible. The MILP takes the advance and
accepts termination. Settlement exactly cancels the advance.

\paragraph{Parameters.}
\[
  n=1,\quad s_1=1,\quad f_1=6,\quad\BAC_1=100,\quad CP_1=100,\quad\pi_1=0.
\]
\[
  M_1=2,\quad\theta=(0.5,\;1.0),\quad\phi=(0.5,\;0.5),\quad
  e_{1,1}=1,\quad e_{1,2}=4.
\]
\[
  \alpha_1=0.30\Rightarrow A_1=30,\quad\delta^{\mathrm{rec}}=0.25,\quad
  \psi=0.10,\quad\rho_1=0.
\]
\[
  \Omega_1=1,\quad\mu_1=0.05,\quad\tau_1^{\mathrm{tol}}=1.
\]
\[
  \bar\eta_{1,1}=1.00,\quad\bar\eta_{1,t}=0.05\text{ for }t\geq2.
\]

Zero spend: counter $1\to0$ at $t=1$, termination at $t=2$.
$P=0$. $R^{\mathrm{term}}=-30$. $Z^*=\gamma^0\cdot30+\gamma^1\cdot(-30)
=30-28.5=\mathbf{1.5}$.

\begin{notebox}
\NOTEH With $\gamma<1$, the advance inflow at $t=1$ is worth more than
the settlement outflow at $t=2$ (discounted). This gives $Z^*=1.5>0$,
unlike the $\gamma=1$ case where $Z^*=0$. The discounting actually
makes the advance harvest slightly profitable even when settlement
cancels it in nominal terms.
\end{notebox}

\paragraph{Assertions for pytest.}
\begin{enumerate}[label=A\arabic*.]
  \item $\sum_t x_{1,t}\leq1.0$.
  \item Project terminates at $t\leq3$.
  \item $R_1^{\mathrm{term}}=-30\pm1$.
  \item $|Z^*-1.5|\leq0.5$.
\end{enumerate}

\subsection{SP-A: Late Recovery Threshold ($\psi=0.40$)}
% ------------------------------------------------------------

\paragraph{Parameters and payment mechanics unchanged from previous

\[
  n=1,\quad\BAC_1=120,\quad\pi_1=0.50\Rightarrow CP_1=180,\quad
  M_1=3,\quad\phi=(1/3,\;1/3,\;1/3).
\]
